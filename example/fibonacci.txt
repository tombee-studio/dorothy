import "stdio.h";

func fib(n: int) {
    if (n == 1) {
        return 1;
    }
    if (n == 2) {
        return 1;
    }
    return fib(n - 1) + fib(n - 2);
}

func main() {
    var fmt = "%d\n";
    var i: int = 1;
    while (i <= 10) {
        var r: int = fib(i);
        printf(fmt, r);
        i = i + 1;
    }
    return 0;
}
